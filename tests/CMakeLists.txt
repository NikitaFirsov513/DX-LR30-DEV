cmake_minimum_required(VERSION 3.20)
project(pj26_host_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
add_executable(test_frames test_frames.c ../app/serial_frames.c)
target_include_directories(test_frames PRIVATE ../app)
add_test(NAME serial_frames COMMAND test_frames)
add_executable(test_radio test_radio.c ../devices/sx126x/radio.c ../vendor/semtech/sx126x.c)
target_include_directories(test_radio PRIVATE fakes ../devices/sx126x ../vendor/semtech)
add_test(NAME radio_state_machine COMMAND test_radio)
add_executable(test_diagnostics test_diagnostics.c ../app/diagnostics.c)
target_include_directories(test_diagnostics PRIVATE ../app ../devices/sx126x)
add_test(NAME radio_diagnostics COMMAND test_diagnostics)
foreach(target test_frames test_radio test_diagnostics)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W3 /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter -UNDEBUG)
    endif()
endforeach()
